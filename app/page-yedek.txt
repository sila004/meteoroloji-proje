
"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import {
  cities,
  getWeather,
  weatherDescription,
  type Weather,
} from "./lib/weather";
import styles from "./features.module.css";

type Forecast = {
  hourly: {
    time: string[];
    temperature_2m: number[];
    precipitation_probability: number[];
  };
  daily: {
    time: string[];
    weather_code: number[];
    temperature_2m_max: number[];
    temperature_2m_min: number[];
    precipitation_probability_max: number[];
  };
};

const pages = [
  { title: "🗺️ Türkiye Haritası", href: "/harita" },
  { title: "⚖️ Şehir Karşılaştırma", href: "/karsilastir" },
  { title: "🌿 Hava Kalitesi", href: "/hava-kalitesi" },
  { title: "❤️ Favorilerim", href: "/favoriler" },
  { title: "📚 Meteoroloji Rehberi", href: "/rehber" },
];

function formatDay(date: string) {
  return new Date(date + "T12:00:00").toLocaleDateString("tr-TR", {
    weekday: "long",
    day: "numeric",
    month: "long",
  });
}

export default function Home() {
  const [selected, setSelected] = useState(0);
  const [weather, setWeather] = useState<Weather | null>(null);
  const [forecast, setForecast] = useState<Forecast | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const city = cities[selected];

  useEffect(() => {
    let active = true;

    async function loadData() {
      setLoading(true);
      setError("");

      try {
        const params = new URLSearchParams({
          latitude: String(city.lat),
          longitude: String(city.lon),
          hourly: "temperature_2m,precipitation_probability",
          daily:
            "weather_code,temperature_2m_max,temperature_2m_min,precipitation_probability_max",
          forecast_days: "7",
          timezone: "auto",
        });

        const [currentData, forecastResponse] = await Promise.all([
          getWeather(city),
          fetch(`https://api.open-meteo.com/v1/forecast?${params}`),
        ]);

        if (!forecastResponse.ok) {
          throw new Error("Tahmin verileri alınamadı.");
        }

        const forecastData: Forecast = await forecastResponse.json();

        if (active) {
          setWeather(currentData);
          setForecast(forecastData);
        }
      } catch {
        if (active) {
          setError("Hava durumu yüklenemedi.");
        }
      } finally {
        if (active) setLoading(false);
      }
    }

    loadData();

    return () => {
      active = false;
    };
  }, [city]);

  const hourlyItems = forecast
    ? forecast.hourly.time
        .map((time, index) => ({
          time,
          temperature: forecast.hourly.temperature_2m[index],
          rain: forecast.hourly.precipitation_probability[index],
        }))
        .filter(
          (item) =>
            new Date(item.time).getTime() >= Date.now() - 3600000
        )
        .slice(0, 12)
    : [];

  return (
    <main className={styles.page}>
      <div className={styles.container}>
        <h1 className={styles.heading}>🌤️ METEOROLOJİ</h1>

        <p className={styles.subtitle}>
          Türkiye şehirlerinin güncel hava durumunu keşfet.
        </p>

        <section className={styles.card}>
          <h2>📍 Şehir Seç</h2>

          <select
            className={styles.select}
            value={selected}
            onChange={(e) => setSelected(Number(e.target.value))}
          >
            {cities.map((item, index) => (
              <option key={item.name} value={index}>
                {item.name}
              </option>
            ))}
          </select>

          {loading && <p>Yükleniyor...</p>}
          {error && <p className={styles.error}>{error}</p>}

          {weather && (
            <>
              <h2>{city.name}</h2>
              <div className={styles.value}>
                {Math.round(weather.temperature)}°C
              </div>
              <h3>{weatherDescription(weather.code)}</h3>

              <div className={styles.grid}>
                <div className={styles.card}>
                  🌡️ Hissedilen: {weather.apparent}°C
                </div>
                <div className={styles.card}>
                  💧 Nem: %{weather.humidity}
                </div>
                <div className={styles.card}>
                  💨 Rüzgâr: {weather.wind} km/sa
                </div>
              </div>
            </>
          )}
        </section>

        <section style={{ marginTop: 36 }}>
          <h2>🕒 Saatlik Hava Durumu</h2>

          <div className={styles.grid}>
            {hourlyItems.map((item) => (
              <div className={styles.card} key={item.time}>
                <h3>{item.time.split("T")[1]}</h3>
                <div className={styles.value}>
                  {Math.round(item.temperature)}°C
                </div>
                <p>🌧️ Yağış: %{item.rain ?? "—"}</p>
              </div>
            ))}
          </div>
        </section>

        <section style={{ marginTop: 36 }}>
          <h2>📅 7 Günlük Tahmin</h2>

          <div className={styles.grid}>
            {forecast?.daily.time.map((day, index) => (
              <div className={styles.card} key={day}>
                <h3>{formatDay(day)}</h3>
                <p>
                  {weatherDescription(
                    forecast.daily.weather_code[index]
                  )}
                </p>
                <div className={styles.value}>
                  {Math.round(
                    forecast.daily.temperature_2m_max[index]
                  )}°C
                </div>
                <p>
                  En düşük:{" "}
                  {Math.round(
                    forecast.daily.temperature_2m_min[index]
                  )}°C
                </p>
              </div>
            ))}
          </div>
        </section>

        <section style={{ marginTop: 36 }}>
          <h2> Diğer Özellikler</h2>

          <div className={styles.grid}>
            {pages.map((page) => (
              <div className={styles.card} key={page.href}>
                <h3>{page.title}</h3>
                <Link href={page.href} className={styles.button}>
                  Sayfaya Git →
                </Link>
              </div>
            ))}
          </div>
        </section>
      </div>
    </main>
  );
}
